package pluginstore
import (
 "os"
 "testing"
)
func TestReserveRegistrySchema(t *testing.T) {
 path:=os.Getenv("RESERVE_REGISTRY");if path=="" {t.Fatal("RESERVE_REGISTRY is required")}
 raw,err:=os.ReadFile(path);if err!=nil {t.Fatal(err)}
 registry,err:=ParseRegistry(raw);if err!=nil {t.Fatal(err)}
 if registry.SchemaVersion!=1 || len(registry.Plugins)!=1 || registry.Plugins[0].ID!="codex-quota-reserve" || PluginInstallType(registry.Plugins[0])!=InstallTypeGitHubRelease {t.Fatalf("registry contract mismatch: %+v",registry)}
}
