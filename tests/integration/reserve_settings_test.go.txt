package management

import (
	"net/http"
	"net/http/httptest"
	"os"
	"strings"
	"testing"

	"github.com/gin-gonic/gin"
	"github.com/router-for-me/CLIProxyAPI/v8/internal/config"
	"gopkg.in/yaml.v3"
)

func TestReserveSettingsPatchPersists(t *testing.T) {
	const id = "codex-quota-reserve"
	h := &Handler{
		cfg: &config.Config{
			Plugins: config.PluginsConfig{
				Configs: map[string]config.PluginInstanceConfig{
					id: pluginConfigFromYAML(t, "enabled: true\nstore:\n  id: codex-quota-reserve\n  version: 0.2.0\n"),
				},
			},
		},
		configFilePath: writeTestConfigFile(t),
	}
	for _, body := range []string{
		`{"reserve_enabled":true,"reserve_5h_percent":15,"reserve_warning_percent":17,"usage_report_enabled":true,"usage_report_minutes":60}`,
		`{"reserve_enabled":false,"reserve_5h_percent":0,"reserve_warning_percent":0,"usage_report_enabled":false,"usage_report_minutes":120}`,
	} {
		rec := httptest.NewRecorder()
		c, _ := gin.CreateTestContext(rec)
		c.Params = gin.Params{{Key: "id", Value: id}}
		c.Request = httptest.NewRequest(http.MethodPatch, "/v0/management/plugins/"+id+"/config", strings.NewReader(body))
		c.Request.Header.Set("Content-Type", "application/json")
		h.PatchPluginConfig(c)
		if rec.Code != http.StatusOK {
			t.Fatalf("PATCH status=%d body=%s", rec.Code, rec.Body.String())
		}
		data, err := os.ReadFile(h.configFilePath)
		if err != nil {
			t.Fatal(err)
		}
		var saved struct {
			Plugins struct {
				Configs map[string]map[string]any `yaml:"configs"`
			} `yaml:"plugins"`
		}
		if err := yaml.Unmarshal(data, &saved); err != nil {
			t.Fatal(err)
		}
		item := saved.Plugins.Configs[id]
		if item["enabled"] != true {
			t.Fatalf("host enabled flag overwritten: %#v", item)
		}
		store, ok := item["store"].(map[string]any)
		if !ok || store["version"] != "0.2.0" {
			t.Fatalf("store metadata overwritten: %#v", item)
		}
		if _, exists := item["webhook_url"]; exists {
			t.Fatal("private Webhook URL appeared in host config")
		}
		if strings.Contains(body, `"reserve_enabled":true`) {
			if item["reserve_enabled"] != true || item["reserve_5h_percent"] != 15 || item["reserve_warning_percent"] != 17 || item["usage_report_enabled"] != true {
				t.Fatalf("saved settings mismatch: %#v", item)
			}
		} else if item["reserve_enabled"] != false || item["reserve_5h_percent"] != 0 || item["reserve_warning_percent"] != 0 || item["usage_report_enabled"] != false || item["usage_report_minutes"] != 120 {
			t.Fatalf("saved zero/false settings mismatch: %#v", item)
		}
	}
}
